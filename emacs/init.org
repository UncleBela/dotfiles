  (let ((bootstrap-file
	 (expand-file-name
	  "straight/repos/straight.el/bootstrap.el"
	  (or (bound-and-true-p straight-base-dir)
	      user-emacs-directory)))
	(bootstrap-version 7))
    (unless (file-exists-p bootstrap-file)
      (with-current-buffer
	  (url-retrieve-synchronously
	   "https://raw.githubusercontent.com/radian-software/straight.el/develop/install.el"
	   'silent 'inhibit-cookies)
	(goto-char (point-max))
	(eval-print-last-sexp)))
    (load bootstrap-file nil 'nomessage))

  ;; Silencing some specific LSP warnings.
  (defun my-filter-lsp-warnings (format-string &rest args)
    "Filter out specific lsp-mode warnings."
    (unless (string-match-p "Unknown request method: workspace/diagnostic/refresh" format-string)
      (apply #'message format-string args)))

  (advice-add 'lsp-warn :override #'my-filter-lsp-warnings)

    "Custom yasnippet selection with ivy from ~/.emacs.d/snippets dir"
    (interactive)
    (let* ((yas-snippet-dirs '("~/.emacs.d/snippets")) ; Default *snippets*
           (choices (yas--all-templates (yas--get-snippet-tables)))
           (my-snippets (seq-filter

      (ivy-read "Snippet: " (mapcar #'car snippets)
                :action (lambda (snippet-name)
                          (let ((template (cdr (assoc snippet-name snippets))))
                            (when template
                              (yas-expand-snippet (yas--template-content template))))))))

#+begin_src emacs-lisp :tangle "gyb-lisp/gyb-org-image-formatting.el" :mkdirp yes
(defun capitalize-first-letter (s)
  "Capitalize the first letter of string S. This is used for my function wrap image. Return the same input S if it's empty or not a string."
  (when (stringp s)
    (cond
     ((string-empty-p s) s)
     (t (concat (upcase (substring s 0 1)) (substring s 1))))))

  (defun my-org-html--wrap-image (contents info &optional label attributes)
        "Don’t wrap images in <figure> or <figcaption>.
      CONTENTS is the HTML for the image itself.
      INFO is the export communication channel.
      LABEL is the label for the environment (if any).
      ATTRIBUTES is a plist of HTML attributes (if any)."
        ;; Simply return the image HTML string as-is:
        contents)

      (advice-add 'org-html--wrap-image :override #'my-org-html--wrap-image)

(defun my-org-html--format-image (source attributes info)
  "Wrap <img> in an <a> tag for fancyBox, using filename-derived text for alt/title/caption.
Also capitalize the very first letter of the resulting text.
SOURCE is the image path.
ATTRIBUTES is a plist for image attributes.
INFO is a plist used as the export communication channel."
  (let* ((file-name    (file-name-nondirectory source))     ; e.g., "great-kitty_photo.png"
         (base-name    (file-name-sans-extension file-name)) ; "great-kitty_photo"
         ;; Replace underscores/hyphens with spaces, e.g. "great kitty photo"
         (raw-text     (replace-regexp-in-string "[-_]+" " " base-name))
         ;; Capitalize only the first letter of the entire string
         (caption-text (capitalize-first-letter raw-text))
         ;; Merge any existing attributes with ours
         (img-attrs    (org-combine-plists
                        (list :src source
                              :alt caption-text
                              :title caption-text)
                        attributes)))
    (concat
     (format "<a href=\"%s\" data-fancybox=\"gallery\" data-caption=\"%s\">"
             source
             (org-html-encode-plain-text caption-text))
     (format "<img %s/>" (org-html--make-attribute-string img-attrs))
     "</a>")))

(advice-add 'org-html--format-image :override #'my-org-html--format-image)
#+end_src
